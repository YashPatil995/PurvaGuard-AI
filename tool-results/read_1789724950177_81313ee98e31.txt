    86→function OverviewTab() {
    87→  const [counts, setCounts] = React.useState<{
    88→    alerts: number; incidents: number; news: number; facilities: number; regions: number; volunteers: number
    89→  } | null>(null)
    90→  const [loading, setLoading] = React.useState(true)
    91→
    92→  React.useEffect(() => {
    93→    let cancelled = false
    94→    ;(async () => {
    95→      setLoading(true)
    96→      try {
    97→        const [alerts, incidents, news, facilities, regions, volunteers] = await Promise.all([
    98→          apiGet<{ items: any[]; total: number }>(`/api/ops/alerts`).catch(() => ({ items: [], total: 0 })),
    99→          apiGet<{ items: any[]; total: number }>(`/api/ops/incidents?limit=500`).catch(() => ({ items: [], total: 0 })),
   100→          apiGet<{ items: any[]; total: number }>(`/api/admin/news`).catch(() => ({ items: [], total: 0 })),
   101→          apiGet<{ items: any[]; total: number }>(`/api/admin/facilities`).catch(() => ({ items: [], total: 0 })),
   102→          apiGet<{ items: any[]; total: number }>(`/api/admin/regions`).catch(() => ({ items: [], total: 0 })),
   103→          apiGet<{ items: any[]; total: number }>(`/api/admin/volunteers`).catch(() => ({ items: [], total: 0 })),
   104→        ])
   105→        if (cancelled) return
   106→        setCounts({
   107→          alerts: alerts.total,
   108→          incidents: incidents.total,
   109→          news: news.total,
   110→          facilities: facilities.total,
   111→          regions: regions.total,
   112→          volunteers: volunteers.total,
   113→        })
   114→      } finally {
   115→        if (!cancelled) setLoading(false)
   116→      }
   117→    })()
   118→    return () => { cancelled = true }
   119→  }, [])
   120→
   121→  const cards = counts
   122→    ? [
   123→        { label: 'Alerts (all statuses)', value: counts.alerts, icon: Newspaper, tone: 'text-red-600 dark:text-red-300' },
   124→        { label: 'Incidents', value: counts.incidents, icon: AlertTriangle, tone: 'text-orange-600 dark:text-orange-300' },
   125→        { label: 'News items', value: counts.news, icon: Newspaper, tone: 'text-blue-600 dark:text-blue-300' },
   126→        { label: 'Facilities', value: counts.facilities, icon: Building2, tone: 'text-emerald-600 dark:text-emerald-300' },
   127→        { label: 'Regions / localities', value: counts.regions, icon: MapPin, tone: 'text-violet-600 dark:text-violet-300' },

   129→      ]
   130→    : []
   131→
   132→  return (
   133→    <div className="space-y-4">
   134→      <div className="flex items-center gap-2 flex-wrap">
   135→        <Badge variant="outline" className="gap-1 border-amber-300 bg-amber-50 text-amber-800 dark:bg-amber-900/30 dark:text-amber-300 dark:border-amber-700">
   136→          <span className="h-1.5 w-1.5 rounded-full bg-amber-500" />
   137→          demo mode: ON
   138→        </Badge>
   139→        <Badge variant="outline" className="gap-1">
   140→          <ShieldCheck className="h-3 w-3" /> Audit logging active
   141→        </Badge>
   142→        <span className="text-xs text-muted-foreground">
   143→          Simulation labels are shown on every simulated object so demo audiences can distinguish synthetic from live data.
   144→        </span>
   145→      </div>

   147→        <div className="grid grid-cols-2 md:grid-cols-3 gap-3">

   153→            <Card key={c.label} className="py-4">

   156→                  <c.icon className="h-5 w-5" />
   157→                </div>

   162→              </CardContent>

   182→function DataHealthInline() {
   183→  const [data, setData] = React.useState<any>(null)
   184→  React.useEffect(() => {
   185→    apiGet<any>(`/api/ops/data-health`)
   186→      .then(setData)
   187→      .catch(() => {})
   188→  }, [])
   189→  if (!data) return <Skeleton className="h-12 w-full" />
   190→  const s = data.freshnessSummary
   191→  return (
   192→    <div className="grid grid-cols-2 sm:grid-cols-5 gap-2 text-xs">
   193→      <Pill label="Total sources" value={s.total} />
   194→      <Pill label="Healthy" value={s.healthy} tone="good" />
   195→      <Pill label="Degraded" value={s.degraded} tone="warn" />
   196→      <Pill label="Errored" value={s.errored} tone="bad" />
   197→      <Pill label="Simulated" value={s.simulated} tone="default" />
   198→    </div>
   199→  )

   202→function Pill({ label, value, tone = 'default' }: { label: string; value: number; tone?: 'default' | 'good' | 'warn' | 'bad' }) {
   203→  const toneCls = {
   204→    default: 'border-border bg-muted/40',
   205→    good: 'border-emerald-200 bg-emerald-50 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-300 dark:border-emerald-800',
   206→    warn: 'border-amber-200 bg-amber-50 text-amber-700 dark:bg-amber-900/30 dark:text-amber-300 dark:border-amber-800',
   207→    bad: 'border-red-200 bg-red-50 text-red-700 dark:bg-red-900/30 dark:text-red-300 dark:border-red-800',
   208→  }[tone]
   209→  return (
   210→    <div className={cn('rounded-md border px-2.5 py-1.5', toneCls)}>
   211→      <div className="text-lg font-bold tabular-nums">{value}</div>
   212→      <div className="text-[10px] uppercase tracking-wide">{label}</div>
   213→    </div>
   214→  )
   215→}

   217→// ─── News tab ────────────────────────────────────────────────────────
   218→function NewsTab({ refreshKey, onRefresh }: { refreshKey: number; onRefresh: () => void }) {
   219→  const [items, setItems] = React.useState<any[]>([])
   220→  const [loading, setLoading] = React.useState(true)
   221→  const [createOpen, setCreateOpen] = React.useState(false)
   222→  const [editItem, setEditItem] = React.useState<any | null>(null)
   223→
   224→  const load = React.useCallback(async () => {
   225→    setLoading(true)
   226→    try {
   227→      const res = await apiGet<{ items: any[] }>(`/api/admin/news`)
   228→      setItems(res.items ?? [])
   229→    } catch (e) {
   230→      toast.error('Failed to load news', { description: String(e) })
   231→    } finally {
   232→      setLoading(false)
   233→    }
   234→  }, [])
   235→
   236→  React.useEffect(() => { load() }, [load, refreshKey])
   237→
   238→  async function togglePin(item: any) {
   239→    try {
   240→      await apiPatch(`/api/admin/news/${item.id}`, { pinned: !item.pinned })
   241→      toast.success(item.pinned ? 'News unpinned' : 'News pinned', { description: 'Action logged.' })
   242→      await load()
   243→      onRefresh()
   244→    } catch (e) {
   245→      toast.error('Failed to toggle pin', { description: String(e) })
   246→    }
   247→  }
   248→
   249→  async function archive(id: string) {
   250→    try {
   251→      const res = await fetch(`/api/admin/news/${id}`, { method: 'DELETE' })
   252→      if (!res.ok) throw new Error(`API ${res.status}`)
   253→      toast.success('News archived', { description: 'Status set to ARCHIVED. Action logged.' })
   254→      await load()
   255→      onRefresh()
   256→    } catch (e) {
   257→      toast.error('Archive failed', { description: String(e) })
   258→    }
   259→  }
   260→
   261→  return (
   262→    <Card className="py-4">
   263→      <CardHeader className="pb-2">
   264→        <div className="flex items-center justify-between flex-wrap gap-2">
   265→          <div>
   266→            <CardTitle className="text-sm flex items-center gap-2">
   267→              <Newspaper className="h-4 w-4" /> News & updates
   268→            </CardTitle>
   269→            <CardDescription className="text-xs">Create, edit, pin, or archive news items.</CardDescription>
   270→          </div>
   271→          <Button size="sm" onClick={() => setCreateOpen(true)}>
   272→            <Plus className="h-3.5 w-3.5" /> New item
   273→          </Button>
   274→        </div>
   275→      </CardHeader>
   276→      <CardContent>
   277→        <div className="rounded-md border max-h-[60vh] overflow-y-auto scrollbar-thin">
   278→          <Table>
   279→            <TableHeader className="sticky top-0 bg-background z-10">
   280→              <TableRow>
   281→                <TableHead className="text-xs">Title</TableHead>
   282→                <TableHead className="text-xs">Publisher</TableHead>
   283→                <TableHead className="text-xs">Categories</TableHead>
   284→                <TableHead className="text-xs">Status</TableHead>
   285→                <TableHead className="text-xs">Pinned</TableHead>
   286→                <TableHead className="text-xs">Published</TableHead>
   287→                <TableHead className="text-xs"></TableHead>
   288→              </TableRow>
   289→            </TableHeader>
   290→            <TableBody>
   291→              {loading && (
   292→                <TableRow><TableCell colSpan={7} className="py-6 text-center text-xs text-muted-foreground">Loading…</TableCell></TableRow>
   293→              )}
   294→              {!loading && items.length === 0 && (
   295→                <TableRow><TableCell colSpan={7} className="py-6 text-center text-xs text-muted-foreground">No news items yet.</TableCell></TableRow>
   296→              )}
   297→              {!loading && items.map((n) => (
   298→                <TableRow key={n.id}>
   299→                  <TableCell className="max-w-[280px]">
   300→                    <div className="text-xs font-medium truncate">{n.title}</div>
   301→                    <div className="text-[10px] text-muted-foreground truncate">{n.summary}</div>
   302→                    {n.demoLabel && <Badge variant="outline" className="mt-0.5 text-[9px] border-amber-300 bg-amber-50 text-amber-800 dark:bg-amber-900/30 dark:text-amber-300">DEMO</Badge>}
   303→                  </TableCell>
   304→                  <TableCell className="text-xs">{n.publisher}</TableCell>
   305→                  <TableCell className="text-xs"><Badge variant="outline" className="text-[10px]">{n.categories}</Badge></TableCell>
   306→                  <TableCell><StatusBadge status={n.status} /></TableCell>
   307→                  <TableCell>
   308→                    <Button size="sm" variant="ghost" className="h-7 w-7 p-0" onClick={() => togglePin(n)} aria-label="Toggle pin">
   309→                      {n.pinned ? <Pin className="h-3.5 w-3.5 text-amber-500" /> : <PinOff className="h-3.5 w-3.5 text-muted-foreground" />}
   310→                    </Button>
   311→                  </TableCell>
   312→                  <TableCell className="text-xs">{n.publishedAt ? formatRelativeTime(n.publishedAt) : '—'}</TableCell>
   313→                  <TableCell>
   314→                    <div className="flex items-center gap-1">
   315→                      <Button size="sm" variant="ghost" className="h-7 w-7 p-0" onClick={() => setEditItem(n)} aria-label="Edit">
   316→                        <Pencil className="h-3.5 w-3.5" />
   317→                      </Button>
   318→                      <AlertDialog>
   319→                        <AlertDialogTrigger asChild>
   320→                          <Button size="sm" variant="ghost" className="h-7 w-7 p-0 text-red-600" aria-label="Archive">
   321→                            <Archive className="h-3.5 w-3.5" />
   322→                          </Button>
   323→                        </AlertDialogTrigger>
   324→                        <AlertDialogContent>
   325→                          <AlertDialogHeader>
   326→                            <AlertDialogTitle>Archive “{n.title}”?</AlertDialogTitle>
   327→                            <AlertDialogDescription>
   328→                              This will soft-delete the news item (status: ARCHIVED). The row will be retained for audit. Bulk actions are not required for this MVP.
   329→                            </AlertDialogDescription>
   330→                          </AlertDialogHeader>
   331→                          <AlertDialogFooter>
   332→                            <AlertDialogCancel>Cancel</AlertDialogCancel>
   333→                            <AlertDialogAction onClick={() => archive(n.id)}>Archive</AlertDialogAction>
   334→                          </AlertDialogFooter>
   335→                        </AlertDialogContent>
   336→                      </AlertDialog>
   337→                    </div>
   338→                  </TableCell>
   339→                </TableRow>
   340→              ))}
   341→            </TableBody>
   342→          </Table>
   343→        </div>
   344→        <p className="text-[11px] text-muted-foreground mt-2">
   345→          Action logged. All mutations create an AuditLog entry authored by the seeded admin user.
   346→        </p>
   347→      </CardContent>
   348→
   349→      <NewsFormDialog
   350→        open={createOpen}
   351→        onOpenChange={setCreateOpen}
   352→        onSaved={async () => { await load(); onRefresh() }}
   353→      />
   354→      <NewsFormDialog
   355→        open={!!editItem}
   356→        onOpenChange={(v) => !v && setEditItem(null)}
   357→        existing={editItem}
   358→        onSaved={async () => { setEditItem(null); await load(); onRefresh() }}
   359→      />
   360→    </Card>
   361→  )
   362→}

   364→function NewsFormDialog({
   365→  open, onOpenChange, existing, onSaved,
   366→}: {
   367→  open: boolean
   368→  onOpenChange: (v: boolean) => void
   369→  existing?: any | null
   370→  onSaved: () => void
   371→}) {
   372→  const [form, setForm] = React.useState({
   373→    title: '', summary: '', publisher: '', sourceUrl: '', categories: 'GENERAL', pinned: false, status: 'PUBLISHED',
   374→  })
   375→  const [saving, setSaving] = React.useState(false)
   376→
   377→  React.useEffect(() => {
   378→    if (existing) {
   379→      setForm({
   380→        title: existing.title ?? '',
   381→        summary: existing.summary ?? '',
   382→        publisher: existing.publisher ?? '',
   383→        sourceUrl: existing.sourceUrl ?? '',
   384→        categories: existing.categories ?? 'GENERAL',
   385→        pinned: existing.pinned ?? false,
   386→        status: existing.status ?? 'PUBLISHED',
   387→      })
   388→    } else if (open) {
   389→      setForm({
   390→        title: '', summary: '', publisher: '', sourceUrl: '', categories: 'GENERAL', pinned: false, status: 'PUBLISHED',
   391→      })
   392→    }
   393→  }, [existing, open])
   394→
   395→  async function save() {
   396→    if (!form.title || !form.summary || !form.publisher || !form.sourceUrl) {
   397→      toast.error('Title, summary, publisher and source URL are required')
   398→      return
   399→    }
   400→    setSaving(true)
   401→    try {
   402→      if (existing) {
   403→        await apiPatch(`/api/admin/news/${existing.id}`, form)
   404→        toast.success('News updated', { description: 'Action logged.' })
   405→      } else {
   406→        await apiPost(`/api/admin/news`, form)
   407→        toast.success('News created', { description: 'Action logged.' })
   408→      }
   409→      onSaved()
   410→      onOpenChange(false)
   411→    } catch (e) {
   412→      toast.error('Failed to save news', { description: String(e) })
   413→    } finally {
   414→      setSaving(false)
   415→    }
   416→  }
   417→
   418→  return (
   419→    <Dialog open={open} onOpenChange={onOpenChange}>
   420→      <DialogContent className="sm:max-w-lg">
   421→        <DialogHeader>
   422→          <DialogTitle>{existing ? 'Edit news item' : 'Create news item'}</DialogTitle>
   423→          <DialogDescription>
   424→            News items are visible in the public News view. Mark as DRAFT to hide, PUBLISHED to show, ARCHIVED to soft-delete.
   425→          </DialogDescription>
   426→        </DialogHeader>
   427→        <div className="space-y-3">
   428→          <div>
   429→            <Label className="text-xs">Title</Label>
   430→            <Input className="h-9 text-sm" value={form.title} onChange={(e) => setForm((f) => ({ ...f, title: e.target.value }))} />
   431→          </div>
   432→          <div>
   433→            <Label className="text-xs">Summary</Label>
   434→            <Textarea className="text-sm min-h-16" value={form.summary} onChange={(e) => setForm((f) => ({ ...f, summary: e.target.value }))} />
   435→          </div>
   436→          <div className="grid grid-cols-2 gap-2">
   437→            <div>
   438→              <Label className="text-xs">Publisher</Label>
   439→              <Input className="h-9 text-sm" value={form.publisher} onChange={(e) => setForm((f) => ({ ...f, publisher: e.target.value }))} />
   440→            </div>
   441→            <div>
   442→              <Label className="text-xs">Categories</Label>
   443→              <Input className="h-9 text-sm" value={form.categories} onChange={(e) => setForm((f) => ({ ...f, categories: e.target.value }))} />
   444→            </div>
   445→          </div>
   446→          <div>
   447→            <Label className="text-xs">Source URL</Label>
   448→            <Input className="h-9 text-sm" value={form.sourceUrl} onChange={(e) => setForm((f) => ({ ...f, sourceUrl: e.target.value }))} />
   449→          </div>
   450→          <div className="grid grid-cols-2 gap-2">
   451→            <div>
   452→              <Label className="text-xs">Status</Label>
   453→              <Select value={form.status} onValueChange={(v) => setForm((f) => ({ ...f, status: v }))}>
   454→                <SelectTrigger className="h-9 text-sm"><SelectValue /></SelectTrigger>
   455→                <SelectContent>
   456→                  <SelectItem value="DRAFT">DRAFT</SelectItem>
   457→                  <SelectItem value="PUBLISHED">PUBLISHED</SelectItem>
   458→                  <SelectItem value="ARCHIVED">ARCHIVED</SelectItem>
   459→                </SelectContent>
   460→              </Select>
   461→            </div>
   462→            <div className="flex items-end gap-2">
   463→              <Button size="sm" variant={form.pinned ? 'default' : 'outline'} onClick={() => setForm((f) => ({ ...f, pinned: !f.pinned }))}>
   464→                {form.pinned ? <Pin className="h-3.5 w-3.5" /> : <PinOff className="h-3.5 w-3.5" />}
   465→                {form.pinned ? 'Pinned' : 'Pin to top'}
   466→              </Button>
   467→            </div>
   468→          </div>
   469→        </div>
   470→        <DialogFooter>
   471→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   472→          <Button onClick={save} disabled={saving}>
   473→            <Save className="h-3.5 w-3.5" /> {existing ? 'Save changes' : 'Create'}
   474→          </Button>
   475→        </DialogFooter>
   476→      </DialogContent>
   477→    </Dialog>
   478→  )
   479→}

   189→   189→function MarkerShape({
   190→   190→  shape,
   191→   191→  color,
   192→   192→  size,
   193→   193→}: {
   194→   194→  shape: MarkerShapeKind
   195→   195→  color: string
   196→   196→  size: number
   197→   197→}) {
   198→   198→  const s = size
   199→   199→  switch (shape) {
   200→   200→    case 'triangle':
   201→   201→      return (
   202→   202→        <svg width={s} height={s} viewBox="0 0 16 18" fill="none" aria-hidden>
   203→   203→          <polygon points="8,0 16,16 0,16" fill={color} stroke="white" strokeWidth="1.5" strokeLinejoin="round" />
   204→   204→          <text x="8" y="13" textAnchor="middle" fontSize="9" fontWeight="bold" fill="white">!</text>
   205→   205→        </svg>
   206→   206→      )
   207→   207→    case 'hexagon':
   208→   208→      return (
   209→   209→        <svg width={s} height={s} viewBox="0 0 16 16" fill="none" aria-hidden>
   210→   210→          <polygon points="8,0 15,4 15,12 8,16 1,12 1,4" fill={color} stroke="white" strokeWidth="1.5" strokeLinejoin="round" />
   211→   211→        </svg>
   212→   212→      )
   213→   213→    case 'circle':
   214→   214→      return (
   215→   215→        <svg width={s} height={s} viewBox="0 0 18 18" fill="none" aria-hidden>
   216→   216→          <circle cx="9" cy="9" r="8" fill={color} stroke="white" strokeWidth="1.5" />
   217→   217→          <circle cx="9" cy="9" r="6" fill="none" stroke="white" strokeWidth="1" opacity="0.6" />
   218→   218→        </svg>
   219→   219→      )
   220→   220→    case 'diamond':
   221→   221→      return (
   222→   222→        <svg width={s} height={s} viewBox="0 0 16 16" fill="none" aria-hidden>
   223→   223→          <polygon points="8,0 16,8 8,16 0,8" fill={color} stroke="white" strokeWidth="1.5" strokeLinejoin="round" />
   224→   224→        </svg>
   225→   225→      )
   226→   226→    case 'square':
   227→   227→      return (
   228→   228→        <svg width={s} height={s} viewBox="0 0 16 18" fill="none" aria-hidden>
   229→   229→          <path d="M2 0 L14 0 L14 12 L8 18 L2 12 Z" fill={color} stroke="white" strokeWidth="1.5" strokeLinejoin="round" />
   230→   230→        </svg>
   231→   231→      )
   232→   232→    case 'dot':
   233→   233→      return (
   234→   234→        <svg width={s} height={s} viewBox="0 0 12 12" fill="none" aria-hidden>
   235→   235→          <circle cx="6" cy="6" r="4.5" fill={color} stroke="white" strokeWidth="1.5" />
   236→   236→        </svg>
   237→   237→      )
   238→   238→  }
   239→   239→}

   132→   132→const RISK_RANK: Record<RiskLevel, number> = { LOW: 0, MODERATE: 1, HIGH: 2, VERY_HIGH: 3 }

   134→   134→function computeOverallRisk(summary: RiskSummaryItem[]): RiskLevel {
   135→   135→  let max: RiskLevel = 'LOW'
   136→   136→  for (const r of summary) {
   137→   137→    if (RISK_RANK[r.level] > RISK_RANK[max]) max = r.level
   138→   138→  }
   139→   139→  return max
   140→   140→}

   142→   142→// Personalized actionable checklist derived from current risk levels + weather.
   143→   143→function generateChecklist(
   144→   144→  summary: RiskSummaryItem[],
   145→   145→  weather: WeatherData | null
   146→   146→): { text: string; tone: 'critical' | 'caution' | 'info' }[] {
   147→   147→  const byHazard: Record<string, RiskSummaryItem | undefined> = {}
   148→   148→  for (const r of summary) byHazard[r.hazard] = r
   149→   149→
   150→   150→  const items: { text: string; tone: 'critical' | 'caution' | 'info' }[] = []
   151→   151→
   152→   152→  const landslide = byHazard['LANDSLIDE']
   153→   153→  if (landslide?.level === 'HIGH' || landslide?.level === 'VERY_HIGH') {
   154→   154→    items.push({ text: 'Move away from steep slopes and the base of hills; watch for new ground cracks.', tone: 'critical' })
   155→   155→  } else if (landslide?.level === 'MODERATE') {
   156→   156→    items.push({ text: 'Stay alert for landslide warning signs: cracks, tilting poles, muddy springs.', tone: 'caution' })
   157→   157→  }
   158→   158→
   159→   159→  const flash = byHazard['FLASH_FLOOD']
   160→   160→  if (flash?.level === 'HIGH' || flash?.level === 'VERY_HIGH') {
   161→   161→    items.push({ text: 'Move to higher ground; avoid river crossings and low-lying areas.', tone: 'critical' })
   162→   162→  } else if (flash?.level === 'MODERATE') {
   163→   163→    items.push({ text: 'Avoid riverbanks and stream crossings until conditions improve.', tone: 'caution' })
   164→   164→  }
   165→   165→
   166→   166→  const rain = byHazard['HEAVY_RAIN']
   167→   167→  if (rain?.level === 'HIGH' || rain?.level === 'VERY_HIGH') {
   168→   168→    items.push({ text: 'Avoid non-essential travel; secure household items and clear drains.', tone: 'critical' })
   169→   169→  } else if (weather && weather.rainfall24hMm > 50) {
   170→   170→    items.push({ text: 'Clear drains and avoid low-lying crossings — recent rainfall is elevated.', tone: 'caution' })
   171→   171→  }
   172→   172→
   173→   173→  const road = byHazard['ROAD_BLOCK']
   174→   174→  if (road?.level === 'HIGH' || road?.level === 'VERY_HIGH') {
   175→   175→    items.push({ text: 'Confirm road status before travel; keep an alternate route in mind.', tone: 'caution' })
   176→   176→  }
   177→   177→
   178→   178→  const quake = byHazard['EARTHQUAKE']
   179→   179→  if (quake?.level === 'HIGH' || quake?.level === 'VERY_HIGH') {
   180→   180→    items.push({ text: 'Review Drop–Cover–Hold On with your family; identify safe spots in each room.', tone: 'caution' })
   181→   181→  }
   182→   182→
   183→   183→  // Always-on general preparedness items (keep this list useful even when low risk)
   184→   184→  items.push({ text: 'Keep your phone charged and a power bank ready.', tone: 'info' })
   185→   185→  items.push({ text: 'Review your family communication plan and emergency contacts.', tone: 'info' })
   186→   186→  items.push({ text: 'Keep your emergency go-bag updated (water, medicines, documents).', tone: 'info' })
   187→   187→
   188→   188→  return items.slice(0, 6)
   189→   189→}

   191→   191→// ── Main view ───────────────────────────────────────────────────────────
   192→   192→export default function HomeView() {
   193→   193→  const { location, setView, connectivity } = useApp()
   194→   194→  const [data, setData] = React.useState<HomePayload | null>(null)
   195→   195→  const [loading, setLoading] = React.useState<boolean>(true)
   196→   196→  const [error, setError] = React.useState<string | null>(null)
   197→   197→
   198→   198→  const load = React.useCallback(() => {
   199→   199→    let cancelled = false
   200→   200→    setLoading(true)
   201→   201→    setError(null)
   202→   202→    const url = `/api/home?lat=${encodeURIComponent(location.lat)}&lng=${encodeURIComponent(location.lng)}&name=${encodeURIComponent(location.name)}`
   203→   203→    apiGet<HomePayload>(url)
   204→   204→      .then((d) => {
   205→   205→        if (cancelled) return
   206→   206→        setData(d)
   207→   207→        setLoading(false)
   208→   208→      })
   209→   209→      .catch((e: unknown) => {
   210→   210→        if (cancelled) return
   211→   211→        const msg = e instanceof Error ? e.message : 'Failed to load home data.'
   212→   212→        setError(msg)
   213→   213→        setLoading(false)
   214→   214→      })
   215→   215→    return () => {
   216→   216→      cancelled = true
   217→   217→    }
   218→   218→  }, [location.lat, location.lng, location.name])
   219→   219→
   220→   220→  React.useEffect(() => load(), [load])
   221→   221→
   222→   222→  return (
   223→   223→    <div className="mx-auto w-full max-w-7xl px-3 sm:px-4 py-4 sm:py-6 space-y-4 md:space-y-6">
   224→   224→      {loading ? (
   225→   225→        <HomeSkeleton />
   226→   226→      ) : error || !data ? (
   227→   227→        <ErrorState message={error ?? 'No data available.'} onRetry={load} />
   228→   228→      ) : (
   229→   229→        <HomeContent data={data} setView={setView} connectivity={connectivity} />
   230→   230→      )}
   231→   231→    </div>
   232→   232→  )
   233→   233→}

   132→const RISK_RANK: Record<RiskLevel, number> = { LOW: 0, MODERATE: 1, HIGH: 2, VERY_HIGH: 3 }
   133→
   134→function computeOverallRisk(summary: RiskSummaryItem[]): RiskLevel {
   135→  let max: RiskLevel = 'LOW'
   136→  for (const r of summary) {
   137→    if (RISK_RANK[r.level] > RISK_RANK[max]) max = r.level
   138→  }
   139→  return max
   140→}

   142→// Personalized actionable checklist derived from current risk levels + weather.
   143→function generateChecklist(
   144→  summary: RiskSummaryItem[],
   145→  weather: WeatherData | null
   146→): { text: string; tone: 'critical' | 'caution' | 'info' }[] {
   147→  const byHazard: Record<string, RiskSummaryItem | undefined> = {}
   148→  for (const r of summary) byHazard[r.hazard] = r
   149→
   150→  const items: { text: string; tone: 'critical' | 'caution' | 'info' }[] = []
   151→
   152→  const landslide = byHazard['LANDSLIDE']
   153→  if (landslide?.level === 'HIGH' || landslide?.level === 'VERY_HIGH') {
   154→    items.push({ text: 'Move away from steep slopes and the base of hills; watch for new ground cracks.', tone: 'critical' })
   155→  } else if (landslide?.level === 'MODERATE') {
   156→    items.push({ text: 'Stay alert for landslide warning signs: cracks, tilting poles, muddy springs.', tone: 'caution' })
   157→  }
   158→
   159→  const flash = byHazard['FLASH_FLOOD']
   160→  if (flash?.level === 'HIGH' || flash?.level === 'VERY_HIGH') {
   161→    items.push({ text: 'Move to higher ground; avoid river crossings and low-lying areas.', tone: 'critical' })
   162→  } else if (flash?.level === 'MODERATE') {
   163→    items.push({ text: 'Avoid riverbanks and stream crossings until conditions improve.', tone: 'caution' })
   164→  }
   165→
   166→  const rain = byHazard['HEAVY_RAIN']
   167→  if (rain?.level === 'HIGH' || rain?.level === 'VERY_HIGH') {
   168→    items.push({ text: 'Avoid non-essential travel; secure household items and clear drains.', tone: 'critical' })
   169→  } else if (weather && weather.rainfall24hMm > 50) {
   170→    items.push({ text: 'Clear drains and avoid low-lying crossings — recent rainfall is elevated.', tone: 'caution' })
   171→  }
   172→
   173→  const road = byHazard['ROAD_BLOCK']
   174→  if (road?.level === 'HIGH' || road?.level === 'VERY_HIGH') {
   175→    items.push({ text: 'Confirm road status before travel; keep an alternate route in mind.', tone: 'caution' })
   176→  }
   177→
   178→  const quake = byHazard['EARTHQUAKE']
   179→  if (quake?.level === 'HIGH' || quake?.level === 'VERY_HIGH') {
   180→    items.push({ text: 'Review Drop–Cover–Hold On with your family; identify safe spots in each room.', tone: 'caution' })
   181→  }
   182→
   183→  // Always-on general preparedness items (keep this list useful even when low risk)
   184→  items.push({ text: 'Keep your phone charged and a power bank ready.', tone: 'info' })
   185→  items.push({ text: 'Review your family communication plan and emergency contacts.', tone: 'info' })
   186→  items.push({ text: 'Keep your emergency go-bag updated (water, medicines, documents).', tone: 'info' })
   187→
   188→  return items.slice(0, 6)
   189→}

   191→// ── Main view ───────────────────────────────────────────────────────────
   192→export default function HomeView() {
   193→  const { location, setView, connectivity } = useApp()
   194→  const [data, setData] = React.useState<HomePayload | null>(null)
   195→  const [loading, setLoading] = React.useState<boolean>(true)
   196→  const [error, setError] = React.useState<string | null>(null)
   197→
   198→  const load = React.useCallback(() => {
   199→    let cancelled = false
   200→    setLoading(true)
   201→    setError(null)
   202→    const url = `/api/home?lat=${encodeURIComponent(location.lat)}&lng=${encodeURIComponent(location.lng)}&name=${encodeURIComponent(location.name)}`
   203→    apiGet<HomePayload>(url)
   204→      .then((d) => {
   205→        if (cancelled) return
   206→        setData(d)
   207→        setLoading(false)
   208→      })
   209→      .catch((e: unknown) => {
   210→        if (cancelled) return
   211→        const msg = e instanceof Error ? e.message : 'Failed to load home data.'
   212→        setError(msg)
   213→        setLoading(false)
   214→      })
   215→    return () => {
   216→      cancelled = true
   217→    }
   218→  }, [location.lat, location.lng, location.name])
   219→
   220→  React.useEffect(() => load(), [load])
   221→
   222→  return (
   223→    <div className="mx-auto w-full max-w-7xl px-3 sm:px-4 py-4 sm:py-6 space-y-4 md:space-y-6">
   224→      {loading ? (
   225→        <HomeSkeleton />
   226→      ) : error || !data ? (
   227→        <ErrorState message={error ?? 'No data available.'} onRetry={load} />
   228→      ) : (
   229→        <HomeContent data={data} setView={setView} connectivity={connectivity} />
   230→      )}
   231→    </div>
   232→  )
   233→}
